cmake_minimum_required(VERSION 3.20)
project(forge_tools C)
if(NOT WIN32)
  message(FATAL_ERROR "forge_tools requires Windows")
endif()
add_library(forge_tools STATIC forge_tools.c)
target_include_directories(forge_tools PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(forge_tools PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
enable_testing()
add_executable(test_forge_tools ../tests/tools/test_tools.c)
target_link_libraries(test_forge_tools PRIVATE forge_tools)
add_test(NAME forge_tools COMMAND test_forge_tools)
