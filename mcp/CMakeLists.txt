cmake_minimum_required(VERSION 3.16)
project(forge_mcp C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "FORGE MCP stdio transport currently requires Windows")
endif()
add_library(forge_mcp forge_mcp.c ../shared/src/forge_json.c ../shared/src/forge_util.c)
target_include_directories(forge_mcp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ../shared/src)
target_compile_definitions(forge_mcp PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(mcp_client_test ../tests/mcp/test_client.c)
target_link_libraries(mcp_client_test PRIVATE forge_mcp)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME mcp_fake_server COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../tests/mcp/test_fake_server.py $<TARGET_FILE:mcp_client_test>)
